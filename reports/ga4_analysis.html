pages_audit <- pages |>
  mutate(
    page_path_original = page_path_and_screen_class,

    has_query_parameter =
      str_detect(page_path_original, fixed("?")),

    page_path_standardised = page_path_original |>
      str_squish() |>

    page_path_standardised = case_when(
      page_path_standardised == "/" ~ "/",
      page_path_standardised == "(not set)" ~ "(not set)",
      TRUE ~ str_remove(page_path_standardised, "/+$")

    page_path_template = page_path_standardised |>
      str_replace_all(
        "(/view-order/)\\d+",
        "\\1{id}"

        "(/view-subscription/)\\d+",

        "(/join-group/)[A-Za-z0-9_-]{16,}",
