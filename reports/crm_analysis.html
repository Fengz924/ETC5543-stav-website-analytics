1 Scope and privacy boundary

This report audits the CRM export CRM20260729.csv. The source contains direct identifiers and sensitive contact information. Individual names, email addresses, telephone numbers, street addresses, IP addresses and free-text notes are used only for validation and are not printed in this report or included in the analysis-ready output.

The raw export remains unchanged in data/raw/CRM/.

2 Data import and CSV structure validation

All 71 source fields are initially imported as character values. This prevents R from silently selecting inconsistent types before the field rules are known.

source_file <- here("data", "raw", "CRM", "CRM20260729.csv")
extract_date <- as.Date("2026-07-29")

crm_import <- suppressWarnings(
  read_csv(
    source_file,
    col_types = cols(.default = col_character()),
    trim_ws = FALSE,
    name_repair = "minimal",
    show_col_types = FALSE,
    progress = FALSE,
    lazy = FALSE
  )
)

3 Record grain and identifier validation

The audit treats each row as one CRM contact record. ID is assessed as the source-system primary key; email and user_id are tested as possible secondary keys but are not assumed to have the same meaning.

normalise_email <- function(x) str_to_lower(str_squish(x))
normalise_phone <- function(x) str_replace_all(x, "[^0-9+]", "")

4 Field groups and data dictionary

field_group <- function(field) case_when(
  field %in% c("id", "full_name", "first_name", "last_name", "email", "user_id", "owner_id") ~ "Identity",
  field %in% c("mobile_phone", "primary_phone", "primary_phone_extension", "street_address_1", "street_address_2", "city", "postal_zip", "region", "country", "time_zone", "ip_address") ~ "Contact and location",
  field %in% c("date_created", "birthday") ~ "Contact dates",
  field %in% c("optin_status", "date_optin_status_changed", "terms_agreement", "gdpr_consent", "gdpr_consent_date", "marketing_consent", "marketing_consent_date", "unsub_date", "unsub_reason", "unsub_feedback") ~ "Consent and subscription",
  field %in% c("lead_source", "source_page", "utm_campaign", "utm_content", "utm_medium", "utm_term", "utm_source") ~ "Acquisition",
  field %in% c("membership_status", "membership_plan", "subscription_start_date", "subscription_end_date", "date_of_cancellation", "reason_for_cancellation", "if_other", "date_retired") ~ "Membership lifecycle",
  field %in% c("year_level", "subjects", "job_title_position", "do_you_hold_an_identified_leadership_role", "do_you_hold_an_identified_leadership_role_other", "primary_teachers_and_schools", "secondary_teachers_and_schools", "p_12_teachers_tertiary_educators_academics_lab_technicians_and_p_12_schools_and_organisations", "your_role") ~ "Professional profile",
  field %in% c("publications_medium", "send_me_all_updates", "join_the_asta_mailing_list", "send_me_specific_updates") ~ "Communication preferences",
  field %in% c("last_sponsorship_purchase", "sponsorships_details_sections", "sponsorship_details_divisions", "supporting_sponsorships") ~ "Sponsorship",
  field %in% c("company_name", "company_website", "company_address", "company_phone", "company_phone_extension", "job_title", "company_department") ~ "Organisation",
  field %in% c("notes", "tags") ~ "Free text and tags",
  TRUE ~ "Other"
)

sensitivity <- function(field) case_when(
  field %in% c("full_name", "first_name", "last_name", "email", "birthday", "mobile_phone", "primary_phone", "street_address_1", "street_address_2", "postal_zip", "ip_address", "notes", "unsub_feedback") ~ "Direct/high-risk personal information",
  field %in% c("id", "user_id", "city", "region", "company_name", "company_address", "company_phone") ~ "Indirect identifier",
  TRUE ~ "Business/analytical field"
)

  )

5 Field-specific placeholder cleaning

No is not cleaned globally. It is retained in genuine yes/no consent fields and converted to missing only in fields where it is a system placeholder.

no_to_na <- function(x) if_else(str_to_lower(x) == "no", NA_character_, x)

crm_work <- crm_work |>
  mutate(across(all_of(no_placeholder_fields), no_to_na))

yes_no_to_logical <- function(x) case_when(
  str_to_lower(x) == "yes" ~ TRUE,
  str_to_lower(x) == "no" ~ FALSE,
  is.na(x) ~ NA,
  TRUE ~ NA
)

crm_work <- crm_work |>
  mutate(

    gdpr_consent_clean = yes_no_to_logical(gdpr_consent),

    email_clean = normalise_email(email),
    email_format_valid = str_detect(email_clean, "^[^[:space:]@]+@[^[:space:]@]+\\.[^[:space:]@]+$")

7 Membership status and plan standardisation

normalise_status <- function(x) {
  key <- x |>
    str_to_lower() |>
    str_squish() |>
    str_replace_all("[ _]", "-")
  case_when(
    key == "active" ~ "active",
    key == "on-hold" ~ "on-hold",
    key == "expired" ~ "expired",
    key == "cancelled" ~ "cancelled",
    key == "pending-cancel" ~ "pending-cancel",
    key == "paused" ~ "paused",
    key == "pending" ~ "pending",
    is.na(key) ~ NA_character_,
    TRUE ~ "review-unmapped"
  )
}

normalise_plan <- function(x) x |>
  str_replace_all("[–—]", "-") |>
  str_replace_all("\\s*-\\s*", " - ") |>
  str_squish()

crm_work <- crm_work |>
  mutate(
    membership_status_clean = normalise_status(membership_status),
    membership_plan_clean = normalise_plan(membership_plan),
    has_membership_record = !is.na(membership_status_clean) | !is.na(membership_plan_clean),
    membership_end_before_start = !is.na(subscription_start_date_clean) & !is.na(subscription_end_date_clean) & subscription_end_date_clean < subscription_start_date_clean,
    cancellation_before_start = !is.na(subscription_start_date_clean) & !is.na(date_of_cancellation_clean) & date_of_cancellation_clean < subscription_start_date_clean,
    active_with_past_end_date = membership_status_clean == "active" & !is.na(subscription_end_date_clean) & subscription_end_date_clean < extract_date,

    unsubscribed_without_reason = !is.na(unsub_date_clean) & is.na(unsub_reason)
  )
